/* ── build both cup layers (we always keep two so they can overlap) ─ */
const stack=document.getElementById('stack');
function makeCup(p){
  const wrap=document.createElement('div');
  wrap.className='cup-wrap';
  const cup=document.createElement('div');
  cup.className='cup';
  cup.style.setProperty('--c-top',p.cup[0]);
  cup.style.setProperty('--c-mid',p.cup[1]);
  cup.style.setProperty('--c-bot',p.cup[2]);
  wrap.appendChild(cup);
  return wrap;
}

const stage    = document.getElementById('stage');
const nameEl   = document.getElementById('s-name');
const notesEl  = document.getElementById('s-notes');
const bandEl   = document.getElementById('wordband');

let current = -1;
let liveCup = null;

function renderNotes(p){
  notesEl.innerHTML='';
  p.notes.forEach(([h,b])=>{
    const d=document.createElement('div');
    d.className='note fade-swap d2';
    d.innerHTML='<h3>'+h+' <em>&#9734;</em></h3><p>'+b+'</p>';
    notesEl.appendChild(d);
  });
}
function fillBand(p){
  // repeated word, static — it swaps text, it does not scroll
  bandEl.textContent = (p.word+'\u00A0\u00A0').repeat(8);
}

function goTo(i, dir){
  if(i===current) return;
  const p=PRODUCTS[i];
  const rev = dir<0;

  // cup handoff
  const incoming=makeCup(p);
  stack.appendChild(incoming);
  if(liveCup && !REDUCED){
    const out=liveCup;
    out.classList.add('is-exiting'); if(rev) out.classList.add('rev');
    out.addEventListener('animationend',()=>out.remove(),{once:true});
  }else if(liveCup){ liveCup.remove(); }

  if(!REDUCED){
    incoming.classList.add('is-entering'); if(rev) incoming.classList.add('rev');
  }
  liveCup=incoming;

  // heading  0s
  nameEl.textContent=p.name;
  nameEl.classList.remove('fade-swap'); void nameEl.offsetWidth; nameEl.classList.add('fade-swap');

  // rail  +0.08s
  railItems.forEach((li,n)=>li.classList.toggle('is-on',n===i));

  // copy  +0.12s
  renderNotes(p);

  // background tint  0.6s crossfade
  stage.style.backgroundColor=p.tint;

  fillBand(p);
  current=i;
}

/* ── scroll scrub for the pinned section ────────────────────────── */
const seriesEl=document.getElementById('series');
seriesEl.style.height=(PRODUCTS.length*100)+'svh';

function updateSeries(){
  const r=seriesEl.getBoundingClientRect();
  const total=seriesEl.offsetHeight-window.innerHeight;
  const p=Math.min(Math.max(-r.top/total,0),0.9999);
  const i=Math.floor(p*PRODUCTS.length);
  if(i!==current) goTo(i, i>current?1:-1);
}

/* ── editorial parallax · left 1 : right 0.2 ────────────────────── */
const edEl=document.getElementById('editorial');
const cols=[...edEl.querySelectorAll('.col')];
function updateParallax(){
  const r=edEl.getBoundingClientRect();
  const vh=window.innerHeight;
  if(r.bottom<-200||r.top>vh+200) return;
  // 0 when section enters, 1 when it leaves
  const p=(vh-r.top)/(vh+r.height);
  cols.forEach(c=>{
    const rate=parseFloat(c.dataset.rate);
    // media column travels far, copy column barely moves → ratio 1 : 0.2
    c.style.transform='translate3d(0,'+(-(p-0.5)*260*rate).toFixed(2)+'px,0)';
  });
}

/* ── velocity-coupled marquee + parallax layer ──────────────────── */
const originEl=document.getElementById('origin');
const typeEl=document.getElementById('originType');
const bgEl=originEl.querySelector('.origin__bg');
let mqX=0, vel=0, lastY=window.scrollY, typeW=0;
function measure(){ typeW=typeEl.scrollWidth/4; }

function tick(){
  const y=window.scrollY;
  const raw=y-lastY; lastY=y;
  vel += (raw-vel)*0.18;                 // smoothed scroll velocity

  const base = REDUCED ? 0 : 0.55;       // px/frame idle drift
  mqX -= base + Math.abs(vel)*0.45;      // faster scroll → faster type
  if(typeW && mqX <= -typeW) mqX += typeW;
  typeEl.style.transform='translate3d('+mqX.toFixed(2)+'px,0,0)';

  // background layer outruns the page (~1.6x) → parallax
  const r=originEl.getBoundingClientRect();
  if(r.bottom>0 && r.top<window.innerHeight){
    const p=(window.innerHeight-r.top)/(window.innerHeight+r.height);
    bgEl.style.transform='translate3d(0,'+((p-0.5)*-90).toFixed(2)+'px,0)';
  }
  requestAnimationFrame(tick);
}

/* ── one scroll listener for the cheap stuff ────────────────────── */
let queued=false;
function onScroll(){
  if(queued) return; queued=true;
  requestAnimationFrame(()=>{ updateSeries(); updateParallax(); queued=false; });
}
addEventListener('scroll',onScroll,{passive:true});
addEventListener('resize',()=>{ measure(); onScroll(); });

document.getElementById('totop').addEventListener('click',()=>{
  scrollTo({top:0,behavior:REDUCED?'auto':'smooth'});
});
